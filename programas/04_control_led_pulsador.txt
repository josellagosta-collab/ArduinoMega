#include <Arduino.h>

const int PULSADOR = 7;
const int LED = 8;

void setup()
{
    pinMode(PULSADOR, INPUT_PULLUP);
    pinMode(LED, OUTPUT);
}

void loop()
{
    int estadoPulsador = digitalRead(PULSADOR);

    if (estadoPulsador == LOW)
    {
        digitalWrite(LED, HIGH);
    }
    else
    {
        digitalWrite(LED, LOW);
    }
}
